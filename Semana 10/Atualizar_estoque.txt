@startuml
top to bottom direction
object ":Gerente" as G
object ":TelaGerente" as T
object ":ControladorProduto" as C
object ":Produto" as P
object ":Catalogo" as Cat

G --> T : 1: selecionarProduto(codigo)
T --> C : 1.1: consultarProduto(codigo)
C --> Cat : 1.1.1: localizarProduto(codigo)
Cat --> C : 1.1.2: retornarProduto(produto)
C --> T : 1.1.3: exibirEstoqueAtual()
T --> G : 1.2: mostrarEstoqueAtual()

G --> T : 2: informarMovimentacao(quantidade, tipo)
T --> C : 2.1: atualizarEstoque(produto, quantidade, tipo)
C --> P : 2.1.1: validarMovimentacao(quantidade, tipo)

C --> P : 2.1.2a [quantidade > 0 e tipo = ENTRADA]: adicionarEstoque(quantidade)
C --> P : 2.1.2b [quantidade > 0 e tipo = SAIDA e estoque suficiente]: retirarEstoque(quantidade)

P --> C : 2.1.3: estoqueAtualizado()
C --> T : 2.2 [sucesso]: exibirConfirmacao()
T --> G : 2.3: mostrarConfirmacao()

C --> T : 2.2b [quantidade <= 0]: exibirErroQuantidadeInvalida()
C --> T : 2.2c [tipo = SAIDA e estoque insuficiente]: exibirErroEstoqueInsuficiente()
T --> G : 2.3b: mostrarErro()

@enduml
