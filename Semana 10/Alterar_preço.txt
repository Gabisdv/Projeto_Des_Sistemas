@startuml
top to bottom direction
object ":Gerente" as G
object ":TelaGerente" as T
object ":ControladorProduto" as C
object ":Produto" as P
object ":Catalogo" as Cat

G --> T : 1: selecionarProduto(codigo)
T --> C : 1.1: consultarProduto(codigo)
C --> Cat : 1.1.1: localizarProduto(codigo)
Cat --> C : 1.1.2: retornarProduto(produto)
C --> T : 1.1.3: exibirDadosProduto()
T --> G : 1.2: informarNovoPreco(preco)
G --> T : 2: confirmarAlteracao()
T --> C : 2.1: alterarPreco(produto, preco)

C --> P : 2.1.1 [produto existe e preco > 0]: atualizarPreco(preco)
P --> C : 2.1.2: precoAtualizado()
C --> T : 2.2 [sucesso]: exibirConfirmacao()
T --> G : 2.3: mostrarConfirmacao()

C --> T : 2.2b [produto não existe ou preco <= 0]: exibirErroDadosInvalidos()
T --> G : 2.3b: mostrarErro()

@enduml
